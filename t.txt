// Обратный проход
public override double[] BackwardPass(double[] errors)
{
    double[] gr_sum = new double[numofneurons + 1];
    // Вычисление градиентных сумм выходного слоя
    for (int j = 0; j < numofneurons; j++)
    {
        double sum = 0;
        for (int k = 0; k < numofprevneurons; k++)
            sum += neurons[k].Weights[j + 1] * errors[j];
        gr_sum[j] = sum;
    }

    for (int i = 0; i < numofneurons; i++) // цикл коррекции синаптических весов
    {
        for (int n = 0; n < numofprevneurons + 1; n++)
        {
            double deltaw;
            if (n == 0) // если порог
                deltaw = momentum * lastdeltawights[i, 0] + learningrate * errors[i];
            else
                deltaw = momentum * lastdeltawights[i, n] + learningrate * neurons[i].Inputs[n - 1] * errors[i];
            
            lastdeltawights[i, n] = deltaw;
            neurons[i].Weights[n] += deltaw; // коррекция весов
        }
    }
    return gr_sum;
}
